🛠️ 1. Core Architecture Blueprint

AuditoriaX solves three complex system engineering issues: double-booking race conditions during seat purchases, robust multi-role capabilities, and live camera-based verification at venues.

Key System Highlights:

  • Atomic Seat Assignment Engine: By utilizing the MongoDB query operators dynamically, multiple attendees hitting purchase on the same seat at the exact microsecond will never experience collision. One transaction atomically claim-checks the seat, and the subsequent transaction safely aborts.
  • Dual Database Handover: If local MongoDB is missing on the client machine, the server dynamically falls back to an isolated mongodb-memory-server so the app operates seamlessly inside any dev box.
  • Time-to-Live Seat Locking: Users lock seats during payment processing. The locks are held safe for 300 seconds (5 minutes) and are atomically cleared by MongoDB backend sweeps.

const express = require('express');

  try {
    const { email, password } = req.body;
    const emailLower = email.trim().toLowerCase();
    const user = await User.findOne({ email: emailLower });
    if (!user || !(await bcrypt.compare(password, user.password))) {
      return res.status(401).json({ error: 'Invalid credentials' });
    }
    const token = jwt.sign(
      { id: user._id, email: user.email, role: user.role },
      process.env.JWT_SECRET,
      { expiresIn: '7d' }
    );
    res.cookie('token', token, { httpOnly: true, maxAge: 7 * 24 * 60 * 60 * 1000 });
    res.json({ email: user.email, role: user.role, college: user.college, permissions: user.permissions });
  } catch (err) {
    res.status(500).json({ error: 'Server error' });
  }
});

// Admin User Management routes
router.get('/users', verifyAdmin, async (req, res) => {
  const users = await User.find({}, '-password');
  res.json(users);
});

router.post('/users/create', verifyAdmin, async (req, res) => {
  const { email, password, role, college, permissions } = req.body;
  const hash = await bcrypt.hash(password, 10);
  const user = await User.create({ email, password: hash, role, college, permissions });
  res.json(user);
});

router.put('/users/:email/role', verifyAdmin, async (req, res) => {
  const user = await User.findOne({ email: req.params.email });

  if (req.body.role) user.role = req.body.role;
  if (req.body.permissions) user.permissions = req.body.permissions;
  await user.save();
  res.json(user);
});

router.delete('/users/:email', verifyAdmin, async (req, res) => {

3. Atomic Transaction Booking Engine (routes/bookings.js)

const express = require('express');
const router = express.Router();
const Booking = require('../models/Booking');
const SeatMap = require('../models/SeatMap');
const SeatLock = require('../models/SeatLock');
const Event = require('../models/Event');
const User = require('../models/User');
const { verifyToken, verifyAdmin, verifyRoles } = require('../middleware/auth');

router.post('/lock', verifyToken, async (req, res) => {
  const { eventId, seat } = req.body;
  const userEmail = req.user.email.toLowerCase();
  
  // Double-booking check on permanents
  const sm = await SeatMap.findOne({ eventId });
  if (sm && sm.seats[seat - 1] === true) {
    return res.status(409).json({ error: 'Already booked' });
  }

  await SeatLock.deleteOne({ eventId, userEmail });
  try {
    await SeatLock.create({ eventId, seat, userEmail });
    res.json({ locked: true, seat });
  } catch (err) {
    res.status(409).json({ error: 'Seat locked by another user' });
  }
});

router.post('/', verifyToken, async (req, res) => {
  const { eventId, seat, txId } = req.body;
  const userEmail = req.user.email.toLowerCase();
  const userRole = req.user.role;

  const event = await Event.findOne({ evtId: eventId });
  const userObj = await User.findOne({ email: userEmail });
  const isSameCollege = userObj && event.college.toLowerCase() === (userObj.college || '').toLowerCase();
  const actualPrice = (userRole === 'student' && !isSameCollege) ? event.price : 0;

  if (actualPrice > 0 && !txId) return res.status(400).json({ error: 'Payment required' });

  // Atomic operation to prevent double booking race condition
  const seatIndex = seat - 1;
  const seatField = `seats.${seatIndex}`;
  const sm = await SeatMap.findOneAndUpdate(
    { eventId, [seatField]: false },
    { $set: { [seatField]: true } },
    { new: true }
  );

  if (!sm) return res.status(409).json({ error: 'Seat has been claimed!' });

  const ticketId = 'AX' + Date.now().toString(36).toUpperCase().slice(-8);
  const booking = await Booking.create({
    ticketId, userEmail, userRole, eventId, eventTitle: event.title,
    auditoriumName: event.auditoriumName, seat, price: actualPrice, txId
  });

  // Emit websocket broadcast
  req.app.get('io').emit('seat_booked', { eventId, seatIndex });
  res.status(201).json(booking);
});

module.exports = router;

4. Razorpay Integration (routes/payments.js)

const express = require('express');
const router = express.Router();
const Razorpay = require('razorpay');
const crypto = require('crypto');
const { verifyToken } = require('../middleware/auth');

const razorpay = new Razorpay({
  key_id: process.env.RAZORPAY_KEY_ID,
  key_secret: process.env.RAZORPAY_KEY_SECRET
});

router.post('/create-order', verifyToken, async (req, res) => {
  const { amount, eventId, seat } = req.body;
  const options = {
    amount: Math.round(amount * 100), // convert to paise
    currency: 'INR',
    receipt: `receipt_${eventId}_${seat}`
  };
  const order = await razorpay.orders.create(options);
  res.json({ orderId: order.id, amount: order.amount, keyId: process.env.RAZORPAY_KEY_ID });
});

router.post('/verify', verifyToken, async (req, res) => {
  const { razorpay_order_id, razorpay_payment_id, razorpay_signature } = req.body;
  const body = razorpay_order_id + '|' + razorpay_payment_id;
  const expectedSignature = crypto
    .createHmac('sha256', process.env.RAZORPAY_KEY_SECRET)
    .update(body)
    .digest('hex');

  if (expectedSignature === razorpay_signature) {
    res.json({ verified: true, paymentId: razorpay_payment_id });
  } else {
    res.status(400).json({ error: 'Signature validation failed' });
  }
});

module.exports = router;

const API = '/api';

async function apiFetch(url, options = {}) {
  const defaultOptions = {
    ...options,
    credentials: 'include',
    headers: { 'Content-Type': 'application/json', ...options.headers }
  };
  const r = await fetch(url, defaultOptions);
  const d = await r.json();
  if (!r.ok) throw new Error(d.error || 'Request failed');
  return d;
}

const api = {
